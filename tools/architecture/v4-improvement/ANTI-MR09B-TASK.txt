2. 任务 A：TableAuthority 回执库只读指标（唯一生产改动之一）
----------------------------------------------------------------
在 engine/src/main/kotlin/com/majiang/authority/TableAuthority.kt 中，
紧接 storedReceipts 声明（L195）之后新增（逐字使用）：

    /** MR-09b 回执库只读测量指标（近似估算，仅用于量级对比） */
    data class ReceiptStoreMetrics(
        val currentEntries: Int,
        val rejectedEntries: Int,
        val recordedEntries: Int,
        val appliedEntries: Int,
        val perSeatEntryCounts: List<Int>,
        val approxRetainedBytes: Long
    )

    /**
     * MR-09b 测量入口：只读遍历 storedReceipts，统计条目构成与近似保留字节。
     * 估算口径（粗粒度，不作绝对内存承诺）：对象头 16B；int/long/引用按
     * 4/8/8B；字符串 24B + 2×length；枚举/可空字段按引用计；载荷对象按
     * 其字段同法累加，不计共享驻留与 JVM 压缩。线程安全：ConcurrentHashMap
     * 弱一致遍历，仅用于权威线程外的诊断读取。
     */
    internal fun receiptStoreMetrics(): ReceiptStoreMetrics {
        var rejected = 0
        var recorded = 0
        var applied = 0
        val perSeat = IntArray(4)
        var bytes = 0L
        for ((key, value) in storedReceipts) {
            perSeat[key.seat]++
            when (value.receipt.status) {
                ReceiptStatus.REJECTED -> rejected++
                ReceiptStatus.RECORDED -> recorded++
                ReceiptStatus.APPLIED -> applied++
            }
            bytes += approxReceiptEntryBytes(key, value)
        }
        return ReceiptStoreMetrics(
            currentEntries = storedReceipts.size,
            rejectedEntries = rejected,
            recordedEntries = recorded,
            appliedEntries = applied,
            perSeatEntryCounts = perSeat.toList(),
            approxRetainedBytes = bytes
        )
    }

    private fun approxReceiptEntryBytes(key: ReceiptKey, value: StoredSubmission): Long {
        var bytes = 16L + 4L + 8L                       // ReceiptKey 头+seat+引用
        bytes += 24L + 2L * key.requestId.length        // requestId 字符串
        bytes += 16L + 8L * 3                           // StoredSubmission 头+固定引用
        val receipt = value.receipt
        bytes += 16L + 8L + 4L + 8L + 8L                // ActionReceipt 头+四字段
        bytes += 24L + 2L * receipt.requestId.length
        receipt.reason?.let { bytes += 24L + 2L * it.length }
        value.choice?.let { bytes += 16L + 8L * 3 + 24L + 2L * it.requestId.length }
        value.tileMarkPayload?.let { bytes += 16L + 8L * 3 + 24L + 2L * it.sessionId.length }
        value.sleeveBloomPayload?.let { bytes += 16L + 8L * 3 }
        value.sleeveBloomChoicePayload?.let { bytes += 16L + 8L * 5 + 24L + 2L * it.offerId.length }
        value.beamSwapPayload?.let { bytes += 16L + 8L * 3 }
        value.beamSwapChoicePayload?.let { bytes += 16L + 8L * 5 + 24L + 2L * it.offerId.length }
        return bytes
    }
注意：核对 NormalizedXxxPayload 的真实字段（本文件 L148-184 已给出，
sessionId/roundId/decisionId/offerId 均为 String）后再定稿；ReceiptStatus
需要 import com.majiang.access.ReceiptStatus（对照本文件既有 import，若已
存在则不重复）。除此之外 TableAuthority.kt 不得有任何其它改动。

3. 任务 B：预览提供器调用指标（另一处生产改动）

PlayerPresentationPreviewProvider.kt 的 object 内新增（逐字使用）：

    /** MR-09b 预览调用测量（内部只读指标；有界防泄漏） */
    private val previewCallCount = java.util.concurrent.atomic.LongAdder()
    private val previewNullRoundCount = java.util.concurrent.atomic.LongAdder()
    private val seenDecisionIdentities = java.util.concurrent.ConcurrentHashMap.newKeySet<String>()
    private var previewIdentityFrozen = false
    private const val PREVIEW_IDENTITY_CAP = 8192   // 若 const 不允许放此位置则改为 private val

    /** MR-09b 预览测量快照 */
    data class PreviewCallMetrics(
        val callCount: Long,
        val nullRoundCount: Long,
        val distinctDecisions: Int,
        val identityFrozen: Boolean,
        val duplicateCalls: Long
    )

    internal fun previewCallMetrics(): PreviewCallMetrics = PreviewCallMetrics(
        callCount = previewCallCount.sum(),
        nullRoundCount = previewNullRoundCount.sum(),
        distinctDecisions = seenDecisionIdentities.size,
        identityFrozen = previewIdentityFrozen,
        duplicateCalls = previewCallCount.sum() - previewNullRoundCount.sum() -
            (if (previewIdentityFrozen) 0L else seenDecisionIdentities.size.toLong())
    )

    /** 仅测试隔离用：清零全部指标。 */
    internal fun resetPreviewCallMetrics() {
        previewCallCount.reset()
        previewNullRoundCount.reset()
        seenDecisionIdentities.clear()
        previewIdentityFrozen = false
    }

    private fun recordPreviewCall(roundId: String?, decisionId: String, controlEpoch: Long) {
        previewCallCount.increment()
        if (roundId == null) {
            previewNullRoundCount.increment()
            return
        }
        if (previewIdentityFrozen) return
        val identity = "$roundId|$decisionId|$controlEpoch"
        if (!seenDecisionIdentities.add(identity) ) { /* 重复即重复计算证据 */ }
        if (seenDecisionIdentities.size >= PREVIEW_IDENTITY_CAP) {
            previewIdentityFrozen = true   // 冻结后 duplicateCalls 不再可靠，快照带标志
        }
    }
并在 createPreview 函数体第一行（val roundId = decision.roundId ?: return null
之前）插入：
        recordPreviewCall(decision.roundId, decision.decisionId, decision.controlEpoch)
注意：data class 与 fun 不能用 const 修饰局部位置——若 PREVIEW_IDENTITY_CAP
写法编译可疑，改为 `private val PREVIEW_IDENTITY_CAP = 8192`（object 内合法）。
decision.roundId/decisionId/controlEpoch 字段名以实读
com.majiang.application.PlayerDecision 为准（先读该文件核对）。除此之外
该文件不得有任何其它改动；createPreview 的既有逻辑一行不动。
